\section{Hill-ligningen, Mathieu-ligningen og Floquet-Monodromi}
George William Hills differensialligning (1886) for en periodisk drivfunksjon $f(t+\pi) = f(t)$ er:
\begin{equation}
\frac{d^2 y}{dt^2} + f(t)y = 0, \qquad f(t) = \theta_0 + 2\sum_{n=1}^{\infty}\theta_n\cos(2nt) + \sum_{m=1}^{\infty}\phi_m\sin(2mt).
\end{equation}
Ved å beholde kun $n=0,1$ og $\phi_m = 0$ fremkommer \textbf{Mathieu-ligningen}:
\begin{equation}
\frac{d^2 y}{dt^2} + \big(a - 2q\cos(2t)\big)y = 0.
\end{equation}
La $M(\pi) \in \operatorname{SL}(2,\mathbb{R})$ være fundamentalløsningens monodromimatrise over én periode $\pi$ ($\det M(\pi) = 1$). Da avgjøres hele fremtidens stabilitet av ett enkelt tall i $P$:
\begin{equation}
\boxed{\big|\operatorname{Tr}\big(M(\pi)\big)\big| < 2 \iff \text{Stabil Floquet-kanal (Kapitza-stabilisering ved } a < 0\text{)}.}
\end{equation}

\section{Gauss' Lov i Euklidsk Rom og Krum Romtid}
Integralt og differensialt over en lukket flate $S = \partial V$:
\begin{equation}
\Phi_E = \oint\!\!\!\oint_S \mathbf{E}\cdot d\mathbf{A} = \frac{Q}{\varepsilon_0} \iff \nabla\cdot\mathbf{E} = \frac{\rho}{\varepsilon_0}, \qquad \nabla\cdot\mathbf{D} = \rho_{\text{f}}.
\end{equation}
I en generelt relativistisk krum romtid med metrikkdeterminant $g = \det(g_{\mu\nu})$ og elektromagnetisk felttensor $F^{\kappa 0}$:
\begin{equation}
\boxed{\Phi_E = c\oint\!\!\!\oint_S F^{\kappa 0}\sqrt{-g}\,dS_\kappa.}
\end{equation}

\section{Ampère--Maxwells Lov, Bernoulli og Bessels Ligning}
\begin{align}

\end{align}

\section{BBGKY-Hierarkiet og Boltzmanns Transportligning}
For et $N$-partikkelsystem kobler BBGKY-hierarkiet $s$-partikkelfordelingen $f_s$ til $(s+1)$-partikkelfordelingen $f_{s+1}$:
\begin{equation}
\left(\frac{\partial}{\partial t} + \mathcal{L}_s\right)f_s(\mathbf{x}_1,\dots,\mathbf{x}_s,t) = \sum_{i=1}^s \int d\mathbf{x}_{s+1}\,\Theta_{i,s+1}\,f_{s+1}(\mathbf{x}_1,\dots,\mathbf{x}_{s+1},t).
\end{equation}
Boltzmann kutter kjeden ved $s=1$ via kollisjonsintegralet og etablerer $H$-teoremet:
\begin{equation}
\frac{\partial f}{\partial t} + \mathbf{v}\cdot\nabla_{\mathbf{x}}f + \frac{\mathbf{F}}{m}\cdot\nabla_{\mathbf{v}}f = \left(\frac{\partial f}{\partial t}\right)_{\text{coll}}, \qquad \frac{dS}{dt} \ge 0.
\end{equation}

\section{Burgers' Sjokkligning, Borda--Carnot, Darcy--Weisbach og Cauchy}
\begin{align}
\frac{\partial u}{\partial t} + u\frac{\partial u}{\partial x} &= \nu\frac{\partial^2 u}{\partial x^2} && \text{(Viskøs Burgers' ligning)} \\

\section{Bures--Wasserstein Anti-Kollaps-Prior ved $\sigma \to 0$}

\end{equation}

\section{Fase-Betinget Dissipativ Port-Hamiltonsk Dynamikk og PIC-Identiteten}
Med kanonisk symplektisk matrise $J^\top = -J$, positiv semidefinitt dissipasjon $R_\theta(c_t) = L_\theta(c_t)L_\theta(c_t)^\top \succeq 0$ og Stiefel-ortonormal inngangsport $G_\theta(c_t)^\top G_\theta(c_t) = I_{d_a}$ er ett Euler-steg ($\Delta t = 1$):
\begin{equation}
\hat{s}_{t+1} = F_\theta(s_t, \bar{a}_t \mid c_t) = s_t + \Delta t\Big[\underbrace{\big(J - R_\theta(c_t)\big)\nabla_s H_\theta(s_t, v_t)}_{\text{passiv flyt}} + \underbrace{G_\theta(c_t)\bar{a}_t}_{\text{kontrollflyt}}\Big].
\end{equation}

\begin{teorem}[Port-Inverse Consistency (PIC) Identiteten]
La den inverse handlingsutlesningen være gitt ved transponatet av inngangsporten:
\begin{equation}
\hat{u}_{t+i} = G_\theta(\hat{c}_{t+i})^\top\left[\frac{s_{t+i+1} - \hat{s}_{t+i}}{\Delta t} - \big(J - R_\theta(\hat{c}_{t+i})\big)\nabla_s H_\theta(\hat{s}_{t+i}, \hat{v}_{t+i})\right].
\end{equation}
Fordi $G_\theta(\hat{c}_{t+i})^\top G_\theta(\hat{c}_{t+i}) = I_{d_a}$, er residualet eksakt lik utrullingsfeilen $\delta_{t+i+1} = s_{t+i+1} - \hat{s}_{t+i+1}$ projisert ned på port-underrommet:
\begin{equation}
\boxed{\hat{u}_{t+i} - \bar{a}_{t+i} = \frac{1}{\Delta t}G_\theta(\hat{c}_{t+i})^\top \delta_{t+i+1}}
\end{equation}
og summen $\mathcal{L}_{\text{pred}} + \mathcal{L}_{\text{PIC}}$ (for $K_{\text{PIC}} = K_p$) reduseres til én enkelt kvadratisk form:
\begin{equation}
\boxed{\mathcal{L}_{\text{pred}} + \mathcal{L}_{\text{PIC}} = \frac{1}{|T|K_p r}\sum_{t\in T}\sum_{i=1}^{K_p} \delta_{t+i}^\top\left(I_r + \frac{r}{d_a \Delta t^2}G_{t+i-1}G_{t+i-1}^\top\right)\delta_{t+i}}
\end{equation}
som vekter feil langs de $d_a$ handlingsport-retningene med den eksakte faktoren $w_\parallel = 1 + \frac{r}{d_a\Delta t^2}$ ($17\times$ på Two-Room, $33\times$ på Reacher, $97\times$ på PushT ved $r=192$, og $7,4\times$ på OGB-Cube).
\end{teorem}
